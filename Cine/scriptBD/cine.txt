create database cine_daw collate utf8mb4_spanish_ci;
grant all on cine_daw.* to daw;

create table Sala
(IdSala  int unsigned primary key,
NomSala varchar(50) not null);

create table Pelicula
(IdPeli char(4) primary key,
NomPeli varchar(50) not null,
duracion int unsigned not null,
genero varchar(40) not null,
precio numeric(4,2) not null constraint CK_PrecioPelicula check (Precio between 3 and 20));

create table Cliente
(DNI char(9) primary key,
NomCliente varchar(40) not null,
Apellido varchar(40) not null,
Correo varchar(50) not null unique,
Contraseña blob(40) not null);

create table Compra
(IdCompra int unsigned auto_increment primary key,
Fecha date not null,
Hora time not null,
PrecioTotal numeric(6, 2) not null constraint ck_PrecioTotal check (PrecioTotal > 0),
DescuentoAplicado numeric(6, 2),
DNI char(9) not null,
constraint FK_DNICliente foreign key (DNI) references Cliente(DNI) on update cascade,
constraint CK_TotalDescuentoApli check (DescuentoAplicado < PrecioTotal));

create table Sesion
(IdSesion char (5) primary key,
Fecha date not null,
HoraInicio time not null,
HoraFin time not null,
Precio numeric (4,2) not null constraint CK_PrecioEntrada check (Precio between 3 and 20),
IdSala  int unsigned not null,
IdPeli char (4) not null,
constraint FK_Id_Sala foreign key (IdSala) references Sala (IdSala) on update cascade,
constraint FK_Id_Peli foreign key (IdPeli) references Pelicula (IdPeli) on update cascade,
constraint UQ_Emision unique (IdSala ,Fecha, HoraInicio));


create table Entrada
(IdEntrada int unsigned auto_increment primary key,
PrecioEntrada numeric (5, 2) not null constraint CK_PrecioPorSesion check (PrecioEntrada between 3 and 999),
Descuento numeric (5,2), 
NumEspectadores int unsigned not null constraint CK_Espectadores check (NumEspectadores between 1 and 5),  
IdCompra  int unsigned not null,
IdSesion char (5) not null,
constraint FK_Id_Compra foreign key (IdCompra) references Compra (IdCompra) on update cascade, 
constraint FK_Id_Sesion foreign key (IdSesion) references Sesion (IdSesion) on update cascade,
constraint CK_DescuentoAplicado check (Descuento < PrecioEntrada));
